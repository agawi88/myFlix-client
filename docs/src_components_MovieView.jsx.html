/**
 * @file MovieView.jsx
 * @description
 * Detailed movie view displayed in a modal.
 *
 * Shows full information about a selected movie, including
 * director and genre details, and displays similar movies
 * based on genre and director.
 */

import React from "react";
import PropTypes from "prop-types";
import { Modal, Button } from "react-bootstrap";
import { FaHeart, FaRegHeart } from "react-icons/fa";
import { SimilarMoviesBySections } from "./SimilarMovies/SimilarMoviesBySections";
import { useParams } from "react-router";
import { Link } from "react-router-dom";

/**
 * MovieView component.
 *
 * Retrieves the selected movie based on the URL parameter
 * and displays detailed information in a modal dialog.
 * Also renders lists of similar movies grouped by genre
 * and director.
 *
 * @param {Object} props Component props
 * @param {boolean} props.isFavorite Indicates whether the movie is favorited
 * @param {Function} props.onHide Closes the movie view modal
 * @param {Array<Object>} props.movies List of all movies
 * @param {Function} props.onMovieClick Handles selection of a similar movie
 *
 * @returns {JSX.Element|null} Movie detail modal or null if movie is not found
 */

export const MovieView = ({ isFavorite, onHide, movies, onMovieClick }) => {
  
    /**
   * Determines the current movie based on the route parameter.
   */
  
  const { movieId } = useParams();
  const movie = movies.find((m) => m.id === movieId);

  if (!movie) return null;

return (

    <Modal
          show={!!movie}
          onHide={() => setMovie(null)}
          size="xl"
          centered
          scrollable
          dialogClassName="modal-90w">
          <Modal.Header>
                <Modal.Title>{movie.title}
                  <Button
                    variant="link"
                    // onClick={() => toggleFavorite(movie.id)}
                    className="ms-3 p-0"
          style={{
            color: isFavorite ? "red" : "gray",
            fontSize: "1.5rem"
          }}
          >
            {isFavorite ? <FaHeart /> : <FaRegHeart />} 
          </Button>
      </Modal.Title>
      <Link to={`/`}>
              <button
                onClick={onHide}
                className="backButton" 
                // style={{cursor: "pointer"}}
        >Back</button>
        </Link>
          </Modal.Header>
      <Modal.Body>
            <div>
      <div>
        <img src={movie.image} alt={movie.title}  />
      </div>
      <div>
        <span><b>Title: </b></span>
        <span>{movie.title}</span>
      </div>
      <div>
        <span><b>Release Year: </b></span>
        <span>{movie.releaseYear}</span>
      </div>
      <div>
        <span><b>Setting: </b></span>
        <span>{movie.setting}</span>
      </div>
      <div>
        <span><b>Description: </b></span>
        <span>{movie.description}</span>
      </div>
        <div>

        <li><span>{movie.director.bio}</span></li>
        <li><span><b>Born: {movie.director.dateOfBirth}</b></span></li>
        <li><span><b>Died: {movie.director.deathYear}</b></span></li>
      </ul>
      </div>
        <div>
          <ul>
        <span><b>Genre: {movie.genre.name}</b></span>
        <li><span>{movie.genre.description}</span></li>

      <div>

      </div>

            <hr />

              title="By Genre"
        movies={movies}
        selectedMovie={movie}
              onMovieClick={onMovieClick}

        <SimilarMoviesBySections

          movies={movies}

          onMovieClick={onMovieClick}

            </Modal>

};

  isFavorite: PropTypes.bool.isRequired,

  onMovieClick: PropTypes.func.isRequired,
};
